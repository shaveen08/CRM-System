import React, { useEffect, useRef, useState } from "react";
import Sidebar from "../components/Sidebar";
import Navbar from "../components/Navbar";

// Recharts
import {
  AreaChart,
  Area,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  ResponsiveContainer,
} from "recharts";

// Icons
import { HugeiconsIcon } from "@hugeicons/react";
import {
  Calendar03Icon,
  Download04Icon,
  IndianRupee,
  SaveMoneyDollarIcon,
  UserCheck01Icon,
  UserGroupIcon,
  UserSwitchIcon,
  UserTime01Icon,
} from "@hugeicons/core-free-icons";

// Redux
import { useSelector } from "react-redux";

const Dashboard = () => {
  const [currentTime, setCurrentTime] = useState("");

  useEffect(() => {
    const handleDate = () => {
      const hours = new Date().getHours();
      if (hours < 12) setCurrentTime("Good Morning");
      else if (hours < 18) setCurrentTime("Good Afternoon");
      else setCurrentTime("Good Evening");
    };
    handleDate();
  }, []);

  // Dashboard Main Data
  const dashboardData = useSelector((state) => state.dashboard.data) || {};

  // Dashboard customer leads
  const data =
    useSelector((state) => state.dashboard.datas?.leads) || [];

  // Date
  const today = new Date();
  const formattedDate = today.toLocaleDateString("en-GB");

  // Date Picker
  const [datePicker, setDatePicker] = useState("");
  const datePickerRef = useRef(null);

  const openPicker = () => {
    if (datePickerRef.current) {
      if (datePickerRef.current.showPicker) {
        datePickerRef.current.showPicker();
      } else {
        datePickerRef.current.focus();
      }
    }
  };

  // Color Icons
  const colors = [
    "bg-red-100 text-red-700 border",
    "bg-green-100 text-green-700 border",
    "bg-blue-100 text-blue-700 border",
    "bg-purple-100 text-purple-700 border",
    "bg-pink-100 text-pink-700 border",
  ];
  return (
    <div className="flex h-screen w-full">
      <Sidebar />

      <div className="flex flex-col w-full ml-60 pt-16 overflow-y-auto">
        {/* Navbar */}
        <Navbar />

        {/* Body */}
        <div className="p-4 flex flex-col gap-4 mt-1">
          {/* Page Header */}
          <div className="flex items-center justify-between w-full">
            {/* Greeting */}
            <section className="flex flex-col gap-1">
              <h1 className="text-2xl text-black font-medium">
                {currentTime}, Bobby
              </h1>
              <p className="text-sm text-gray-400">{formattedDate}</p>
            </section>

            {/* Date picker + Export */}
            <section className="relative flex flex-row gap-2">
              {/* Date picker trigger */}
              <div onClick={openPicker} className="cursor-pointer">
                <div className="h-full flex gap-1 items-center justify-center font-medium border border-gray-300 rounded-lg overflow-hidden">
                  <div className="p-3 bg-gray-50 border-r border-gray-300 flex items-center justify-center">
                    <HugeiconsIcon icon={Calendar03Icon} size={18} />
                  </div>
                  <div className="px-4 text-sm text-gray-600">
                    {datePicker || "Select Date"}
                  </div>
                </div>
                <input
                  type="date"
                  ref={datePickerRef}
                  value={datePicker}
                  onChange={(e) => setDatePicker(e.target.value)}
                  className="absolute inset-0 opacity-0 pointer-events-none"
                />
              </div>

              {/* Export button */}
              <button
                type="button"
                className="h-full px-10 py-3 flex gap-2 items-center justify-center text-sm font-medium bg-purple-600 hover:bg-purple-500 active:bg-purple-700 text-white rounded-lg cursor-pointer transition-colors"
              >
                <HugeiconsIcon
                  icon={Download04Icon}
                  size={16}
                  strokeWidth={2.5}
                />
                Export
              </button>
            </section>
          </div>

          {/* ── Stat Cards ── */}
          <section className="flex flex-col gap-3">
            {/* 4-column summary cards */}
            <div className="grid grid-cols-4 gap-3">
              {/* Total Leads */}
              <section className="bg-white border border-gray-100 rounded-xl p-3 flex flex-col gap-3 shadow-lg shadow-gray-100">
                <div className="flex items-center justify-between">
                  <div className="h-8 w-8 bg-purple-50 flex items-center justify-center rounded-lg">
                    <HugeiconsIcon
                      icon={UserGroupIcon}
                      size={16}
                      strokeWidth={1.5}
                      color="#534AB7"
                    />
                  </div>
                  <span className="text-[14px] font-medium text-purple-700 bg-purple-50 px-2.5 py-1 rounded-full">
                    +12 this week
                  </span>
                </div>
                <div>
                  <p className="text-[14px] text-gray-400 mb-0.5">
                    Total Leads
                  </p>
                  <p className="text-2xl font-semibold tracking-tight text-gray-900">
                    {dashboardData?.summary?.totalLeads ?? "—"}
                  </p>
                </div>
                <div className="h-px bg-gray-100" />
                <p className="text-[14px] text-gray-400 flex items-center gap-1">
                  Recent:
                  <span className="text-gray-700 font-medium">
                    {dashboardData?.weeklyActivity?.length || 0}
                  </span>
                </p>
              </section>

              {/* Pending Leads */}
              <section className="bg-white border border-gray-100 rounded-xl p-3 flex flex-col gap-3 shadow-lg shadow-gray-100">
                <div className="flex items-center justify-between">
                  <div className="h-8 w-8 bg-amber-50 flex items-center justify-center rounded-lg">
                    <HugeiconsIcon
                      icon={UserTime01Icon}
                      size={16}
                      strokeWidth={1.5}
                      color="#854F0B"
                    />
                  </div>
                  <span className="text-[14px] font-medium text-amber-700 bg-amber-50 px-2.5 py-1 rounded-full">
                    This month
                  </span>
                </div>
                <div>
                  <p className="text-[14px] text-gray-400 mb-0.5">
                    Pending Leads
                  </p>
                  <p className="text-2xl font-semibold tracking-tight text-gray-900">
                    {dashboardData?.summary?.pendingLeads ?? "—"}
                  </p>
                </div>
                <div className="h-px bg-gray-100" />
                <p className="text-[14px] text-gray-400 flex items-center gap-1">
                  Month pending:
                  <span className="text-gray-700 font-medium">
                    {dashboardData?.monthlyLeads?.[0]?.pending || 0}
                  </span>
                </p>
              </section>

              {/* Completed Leads */}
              <section className="bg-white border border-gray-100 rounded-xl p-3 flex flex-col gap-3 shadow-lg shadow-gray-100">
                <div className="flex items-center justify-between">
                  <div className="h-8 w-8 bg-green-50 flex items-center justify-center rounded-lg">
                    <HugeiconsIcon
                      icon={UserCheck01Icon}
                      size={16}
                      strokeWidth={1.5}
                      color="#3B6D11"
                    />
                  </div>
                  <span className="text-[14px] font-medium text-green-700 bg-green-50 px-2.5 py-1 rounded-full">
                    Closed
                  </span>
                </div>
                <div>
                  <p className="text-[14px] text-gray-400 mb-0.5">
                    Completed Leads
                  </p>
                  <p className="text-2xl font-semibold tracking-tight text-gray-900">
                    {dashboardData?.summary?.completedLeads ?? "—"}
                  </p>
                </div>
                <div className="h-px bg-gray-100" />
                <p className="text-[14px] text-gray-400 flex items-center gap-1">
                  This month:
                  <span className="text-gray-700 font-medium">
                    {dashboardData?.monthlyLeads?.[0]?.completed || 0}
                  </span>
                </p>
              </section>

              {/* Conversion Rate */}
              <section className="bg-white border border-gray-100 rounded-xl p-3 flex flex-col gap-3 shadow-lg shadow-gray-100">
                <div className="flex items-center justify-between">
                  <div className="h-8 w-8 bg-blue-50 flex items-center justify-center rounded-lg">
                    <HugeiconsIcon
                      icon={UserSwitchIcon}
                      size={16}
                      strokeWidth={1.5}
                      color="#185FA5"
                    />
                  </div>
                  <span className="text-[14px] font-medium text-blue-700 bg-blue-50 px-2.5 py-1 rounded-full">
                    By source
                  </span>
                </div>
                <div>
                  <p className="text-[14px] text-gray-400 mb-0.5">
                    Conversion Rate
                  </p>
                  <p className="text-2xl font-semibold tracking-tight text-gray-900">
                    {dashboardData?.summary?.conversionRate ?? "—"}%
                  </p>
                </div>
                <div className="h-px bg-gray-100" />
                <p className="text-[14px] text-gray-400 flex items-center gap-1">
                  Sources:
                  <span className="text-gray-700 font-medium truncate">
                    {dashboardData?.conversionBySource
                      ?.map((s) => s.source)
                      .join(", ") || "N/A"}
                  </span>
                </p>
              </section>

              {/* Revenue Card */}
              <section className="bg-white border col-span-4 border-gray-100 rounded-xl p-3 shadow-lg shadow-gray-100">
                <div className="flex items-start justify-between mb-3">
                  <div className="flex items-center gap-2.5">
                    <div className="h-8 w-8 bg-green-50 flex items-center justify-center rounded-lg">
                      <HugeiconsIcon
                        icon={SaveMoneyDollarIcon}
                        size={16}
                        strokeWidth={1.5}
                        color="#3B6D11"
                      />
                    </div>
                    <div>
                      <p className="text-[14px] text-gray-400">Total Revenue</p>
                      <p className="text-xl font-semibold tracking-tight text-gray-900 flex items-center gap-0.5">
                        <HugeiconsIcon icon={IndianRupee} size={18} />
                        {dashboardData?.summary?.revenueGenerated ?? "—"}
                      </p>
                    </div>
                  </div>
                </div>

                {/* Legend */}
                <div className="flex items-center gap-4 mb-3">
                  <span className="flex items-center gap-1.5 text-xs text-gray-400">
                    <span className="w-2 h-2 rounded-full bg-purple-600 inline-block" />
                    Revenue
                  </span>
                  <span className="flex items-center gap-1.5 text-xs text-gray-400">
                    <span className="w-2 h-2 rounded-full border border-dashed border-green-500 inline-block" />
                    Target
                  </span>
                </div>

                {/* Area Chart */}
                <ResponsiveContainer width="100%" height={180}>
                  <AreaChart
                    data={dashboardData?.revenueByMonth || []}
                    margin={{ top: 4, right: 4, left: -20, bottom: 0 }}
                  >
                    <defs>
                      <linearGradient
                        id="revenueGradient"
                        x1="0"
                        y1="0"
                        x2="0"
                        y2="1"
                      >
                        <stop
                          offset="5%"
                          stopColor="#7851A9"
                          stopOpacity={0.15}
                        />
                        <stop
                          offset="95%"
                          stopColor="#7851A9"
                          stopOpacity={0}
                        />
                      </linearGradient>
                      <linearGradient
                        id="targetGradient"
                        x1="0"
                        y1="0"
                        x2="0"
                        y2="1"
                      >
                        <stop
                          offset="5%"
                          stopColor="#22C55E"
                          stopOpacity={0.1}
                        />
                        <stop
                          offset="95%"
                          stopColor="#22C55E"
                          stopOpacity={0}
                        />
                      </linearGradient>
                    </defs>
                    <CartesianGrid
                      strokeDasharray="3 3"
                      stroke="#f3f4f6"
                      vertical={false}
                    />
                    <XAxis
                      dataKey="month"
                      tick={{ fontSize: 11, fill: "#9ca3af" }}
                      axisLine={false}
                      tickLine={false}
                    />
                    <YAxis
                      tick={{ fontSize: 11, fill: "#9ca3af" }}
                      axisLine={false}
                      tickLine={false}
                    />
                    <Tooltip
                      contentStyle={{
                        background: "#fff",
                        border: "0.5px solid #e5e7eb",
                        borderRadius: "8px",
                        fontSize: "12px",
                        boxShadow: "0 4px 12px rgba(0,0,0,0.06)",
                      }}
                      formatter={(val) => `₹${val.toLocaleString("en-IN")}`}
                    />
                    <Area
                      type="monotone"
                      dataKey="revenue"
                      stroke="#7851A9"
                      fill="url(#revenueGradient)"
                      strokeWidth={1.5}
                      name="Revenue"
                      dot={false}
                      activeDot={{ r: 4, strokeWidth: 0, fill: "#7851A9" }}
                    />
                    <Area
                      type="monotone"
                      dataKey="target"
                      stroke="#22C55E"
                      fill="url(#targetGradient)"
                      strokeWidth={1.5}
                      strokeDasharray="5 4"
                      name="Target"
                      dot={false}
                      activeDot={{ r: 4, strokeWidth: 0, fill: "#22C55E" }}
                    />
                  </AreaChart>
                </ResponsiveContainer>

                <div className="flex items-center justify-start gap-2 pt-3 py-1 border-t border-gray-200">
                  <p className="text-[14px] text-gray-400">
                    This month completed:
                  </p>
                  <p className="text-sm font-medium text-gray-800">
                    {dashboardData?.monthlyLeads?.[0]?.completed ?? "—"} deals
                  </p>
                </div>
              </section>
            </div>
          </section>
        </div>
      </div>
    </div>
  );
};

export default Dashboard;
